how to know if the point on the right
is really on the right side


   +----------------+
           (100, 10)
 
        (50, 5)    (75, 5)

     (0, 0)
   +----------------+

one possibility is to map
one interval onto the other one

we already have the y value
so we can map the y interval
onto the x interval

then we only have to compare the
x values to see if the point
is really on the right side

let interv_map ys xs (cy) =
  let ay, by = (ys) in
  let ax, bx = (xs) in
  let y_diff = (by - ay) in
  let x_diff = (bx - ax) in
  let ry = (cy - ay) in
  ((ry * x_diff) / y_diff) + ax

let linear_mp (ax, ay) (bx, by) (cx, cy) =
  let interv_y = (ay, by) in
  let interv_x = (ax, bx) in
  let r = interv_map interv_y interv_x (cy) in
  let dx, dy = r, cy in
  (dx, dy)

let () =
  let ax, ay = (0, 0) in
  let bx, by = (100, 10) in
  let cx, cy = (75, 5) in
  let dx, dy = linear_mp (ax, ay) (bx, by) (cx, cy) in
  Printf.printf "# %d %d\n%!" dx dy;
;;

let linear_ri (ax, ay) (bx, by) (cx, cy) =
  if ay = by then Stdlib.compare ay cy else
  let (dx, dy) = linear_mp (ax, ay) (bx, by) (cx, cy) in
  Stdlib.compare cx dx

let linear_ri (ax, ay) (bx, by) (cx, cy) =
  if ax = bx && ay = by then -2 else
  linear_ri (ax, ay) (bx, by) (cx, cy)

let () =
  let ax, ay = (0, 0) in
  let bx, by = (100, 10) in
  let cx, cy = (75, 5) in
  let d = linear_ri (ax, ay) (bx, by) (cx, cy) in
  Printf.printf "# %d\n%!" d;
;;
