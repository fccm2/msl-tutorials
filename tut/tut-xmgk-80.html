<msl>
  <masks>
    <layer name='mask1'>
      <new w='200' h='200' rgb='255,255,255' />
      <draw_rectangle pos='0,0' dims='100,100' rgb='0,0,0' />
      <display />
    </layer>
  </masks>
  <image>
    <merge>
      <layer>
        <new w='200' h='200' rgb='255,255,255' />
      </layer>
      <layer mask='mask1'>
        <new w='200' h='200' rgb='25,55,196' />
        <noise attenuate='3.4' />
        <blur radius='0.1' sigma='0.2' />
        <display />
      </layer>
    </merge>
    <display />
  </image>
</msl>
