xmgk - tutorial

here is a tutorial about xmgk, a command-line utility, with which you may eventually, create images

here is a small example, img-1.xml, it will create a small image of dimensions 170x110, with a background color of 195,254,215,

<msl>
  <image>
    <new w='170' h='110' rgb='195,254,215' />
    <display />
  </image>
</msl>

if you add the following line after the display instruction, it will produce the image above,

    <write filename='img-1.png' />

a second example

img-2.xml

<?xml version="1.0" ?>
<msl>
  <image>
    <merge>
      <layer>
        <new w='190' h='230' rgb='195,254,215' />
      </layer>
      <layer compose='multiply'>
        <new w='500' h='400' rgb='255,255,255' />
        <path coord='relative' rgb='195,255,185'>
          <move_to p='55 25' />
	  <quad_curve_to h='29 2' p='31 41' />
	  <quad_curve_to h='-2 30' p='-21 43' />
	  <quad_curve_to h='-3 1' p='-7 2' />
	  <quad_curve_to h='-20 -2' p='-21 -41' />
	  <quad_curve_to h='-1 -39' p='17 -45' />
          <close_path />
        </path>
      </layer>
      <layer compose='multiply'>
        <new w='500' h='400' rgb='255,255,255' />
        <path coord='relative' rgb_stroke='50,110,70' stroke_width='2.8'>
          <move_to p='51 29' />
	  <quad_curve_to h='29 2' p='31 41' />
	  <quad_curve_to h='-2 30' p='-21 43' />
	  <quad_curve_to h='-3 1' p='-7 2' />
	  <quad_curve_to h='-20 -2' p='-21 -41' />
	  <quad_curve_to h='-1 -39' p='13 -39' />
        </path>
      </layer>
    </merge>
    <display />
    <write filename='img-3.png' />
  </image>
</msl>
