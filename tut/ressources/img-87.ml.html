let over_layer cl =
  let ng = Xmgk.negate in
  let l1 = Xmgk.new_layer "lighten" in
  let l1 = Xmgk.layer_add_prim (New(500, 400, (255, 255, 255))) l1 in
  let l1 =
    List.fold_left (fun l1 ((x, y), r) ->
      let cr = Xmgk.draw_circle (x, y) r (0, 0, 0) in
      let l1 = Xmgk.layer_add_filter cr l1 in
      l1
    ) l1 cl
  in
  let l1 = Xmgk.layer_add_filter (Xmgk.blur 1.1 1.2) l1 in
  let l1 = Xmgk.layer_add_filter ng l1 in
  let l1 = Xmgk.close_layer l1 in
  l1
let blured_layer cl =
  let xd, yd = (-2, 2) in
  let ls = Xmgk.new_layer_prim () in
  let ls = Xmgk.layer_add_prim (New(500, 400, (255, 255, 255))) ls in
  let ls =
    List.fold_left (fun ls ((x, y), r) ->
      let cs = Xmgk.draw_circle (x+xd, y+yd) r (0, 0, 105) in
      let ls = Xmgk.layer_add_filter cs ls in
      ls
    ) ls cl
  in
  let ls = Xmgk.layer_add_filter (Xmgk.blur 7.5 10.2) ls in
  let ls = Xmgk.close_layer ls in
  ls
let build_prim d =
  let rec aux_prim d ac =
    if d <= 0 then List.rev(ac) else
    let x, y = 5 + Random.int (500-2*5), 5 + Random.int (400-2*5) in
    let c = 5 + Random.int (65) in
    aux_prim (d-1) (((x, y), c)::ac)
  in
  aux_prim d []
