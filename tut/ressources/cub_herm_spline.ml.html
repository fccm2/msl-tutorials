module CbHrmSpln = Cubherm.CubicHermiteSpline

let plot (x, y) =
  let x = int_of_float x in
  let y = int_of_float y in
  Printf.printf "<draw_circle c='%d,%d' r='1' rgb='90,90,255' />\n" x y

let circ r (x, y) =
  let x = int_of_float x in
  let y = int_of_float y in
  Printf.printf "<draw_circle c='%d,%d' r='%d' rgb='110,110,255' />\n" x y r


let line (x1, y1) (x2, y2) =
  let x1 = int_of_float x1 in
  let y1 = int_of_float y1 in
  let y2 = int_of_float y2 in
  let x2 = int_of_float x2 in
  Printf.printf "<path coord='relative' rgb='210,210,255'>\n" ;
  Printf.printf "  <move_to p='%d,%d' />\n" x1 y1;
  Printf.printf "  <line_to p='%d,%d' />\n" x2 y2;
  (* Printf.printf "  <close_path />\n" ; *)
  Printf.printf "</path>\n" ;
;;

let rec drop_last = function
  | [_] | [] -> []
  | x::xs -> x :: drop_last xs

let point2d_rand (x1, y1) (x2, y2) =
  let x = x1 +. Random.float (x2 -. x1)
  and y = y1 +. Random.float (y2 -. y1) in (x, y)

let m () =

  Printf.printf "<layer>\n";
  Printf.printf "<new w='%d' h='%d' rgb='210,210,255' />\n" 600 400;
  Printf.printf "</layer>\n";
  (*
  let p1 = (20., 40.) in
  let p2 = (80., 40.) in
  let p3 = (140., 280.) in
  let p4 = (240., 140.) in
  let p5 = (280., 240.) in
  let p6 = (380., 240.) in
  let ps = ref [p1; p2; p3; p4; p5; p6] in
  *)
  Random.self_init ();
  let p1 = point2d_rand (20., 20.) (580., 380.) in
  let p2 = point2d_rand (20., 20.) (580., 380.) in
  let p3 = point2d_rand (20., 20.) (580., 380.) in
  let p4 = point2d_rand (20., 20.) (580., 380.) in
  let p5 = point2d_rand (20., 20.) (580., 380.) in
  let p6 = point2d_rand (20., 20.) (580., 380.) in
  let p7 = point2d_rand (20., 20.) (580., 380.) in
  let ps = ref [p1; p2; p3; p4; p5; p6; p7] in

  let rec aux f = function
    | p1::((p2::p3::p4::_) as tail) -> f p1 p2 p3 p4; aux f tail
    | _ -> ()
  in
  let n = 60 in
  for j = 0 to pred 10 do
    Printf.printf "<!-- loop %d -->\n" j;

    Printf.printf "<layer compose='multiply'>\n" ;
    Printf.printf "<new w='%d' h='%d' rgb='255,255,255' />\n" 600 400;

    List.iter (circ 3) !ps;
    aux (fun p1 p2 p3 p4 ->
      line p1 p2;
      line p3 p4;
    ) !ps;
    for i = 0 to n do
      let t = float i /. float n in
      aux (fun p1 p2 p3 p4 ->
        let p =
          CbHrmSpln.pnt 0.0
            ~p_pred:p1 ~p0:p2
            ~p_succ:p4 ~p1:p3
            t
        in
        plot p;
      ) !ps;
    done;
    let p = point2d_rand (20., 20.) (580., 380.) in
    ps := p :: (drop_last !ps);
    Printf.printf "</layer>\n" ;
    Printf.printf "\n";
  done

let () =
  Printf.printf "<msl>\n";
  Printf.printf "<image>\n";
  Printf.printf "<merge>\n";
  m () ;
  Printf.printf "</merge>\n";
  Printf.printf "</image>\n";
  Printf.printf "<display />\n";
  Printf.printf "</msl>\n";
;;
