module Xmgk : sig
  type rgb = int * int * int
  type coord = Relative | Absolute
  type stroke = rgb * float
  type path_new = Path of coord * rgb option * stroke option

  type image_uninitialized
  type image
  type image_closed
  type prim = New of int * int * rgb

  val new_image : unit -> image_uninitialized
  val close_image : image -> image_closed

  val image_add_prim : prim -> image_uninitialized -> image

  type filter
  val image_add_filter : filter -> image -> image

  val display : filter
  val write : string -> filter

  type msl
  type msl_closed

  val new_msl : unit -> msl
  val close_msl : msl -> msl_closed

  val msl_add_image : image_closed -> msl -> msl
  val msl_print : msl_closed -> unit

  type path
  val new_path : path_new -> path
  val path_close : path -> filter

  type path_elem =
    | Move_to of int * int
    | H_line_to of int
    | V_line_to of int
    | Quad_curve_to of (int * int) * (int * int)
    | Close_path

  val path_add_elem : path_elem -> path -> path

end = struct
  type rgb = int * int * int
  type coord = Relative | Absolute
  type stroke = rgb * float
  type path_new = Path of coord * rgb option * stroke option
  type path = string list
  type filter = string list

  type prim = New of int * int * rgb

  type image_uninitialized = string list
  type image = string list
  type image_closed = string list

  type msl = string list
  type msl_closed = string list
  let msl_print msl =
    List.iter print_endline msl

  let msl_add_image image msl =
    image @ msl

  let image_add_filter filter image =
    filter @ image

  let new_msl () =
    "<msl>"::[]
  let close_msl msl =
    "</msl>"::msl |> List.rev

  let new_image () =
    "<image>"::[]
  let close_image image =
    "</image>"::image

  let path_close path = "</path>"::path

  let image_add_prim prim image =
    match prim with
    | New (w, h, (r, g, b)) ->
        let prim = Printf.sprintf "<new w='%d' h='%d' rgb='%d,%d,%d' />" w h r g b in
        prim :: image

  let new_path path_init =
    let s_rgb rgb =
      match rgb with None -> ""
      | Some (r, g, b) ->
          let rgb = Printf.sprintf " rgb='%d,%d,%d'" r g b in
          rgb
    in
    let s_stroke stroke =
      match stroke with None -> ""
      | Some ((r, g, b), sw) ->
          let stroke = Printf.sprintf " rgb_stroke='%d,%d,%d' stroke_width='%g'" r g b sw in
          stroke
    in
    match path_init with
    | Path (coord, rgb, stroke) ->
        let path =
          Printf.sprintf
            "<path coord='relative'%s%s>"
            (s_rgb rgb)
            (s_stroke stroke)
        in
        path :: []

  type path_elem =
    | Move_to of int * int
    | H_line_to of int
    | V_line_to of int
    | Quad_curve_to of (int * int) * (int * int)
    | Close_path

  let path_add_elem elem path =
    match elem with
    | Close_path ->
        let close_path = Printf.sprintf "<close_path />" in
        close_path :: path
    | Move_to (x, y) ->
        let move_to = Printf.sprintf "<move_to p='%d,%d' />" x y in
        move_to :: path
    | H_line_to h ->
        let h_line = Printf.sprintf "<h_line_to h='%d' />" h in
        h_line :: path
    | V_line_to v ->
        let v_line = Printf.sprintf "<v_line_to v='%d' />" v in
        v_line :: path
    | Quad_curve_to ((h0, h1), (p0, p1)) ->
        let quad_curve = Printf.sprintf "<quad_curve_to h='%d,%d' p='%d,%d' />" h0 h1 p0 p1 in
        quad_curve :: path

  let display = Printf.sprintf "<display />" :: []
  let write filename = Printf.sprintf "<write filename='%s' />" filename :: []
end
