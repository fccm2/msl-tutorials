(** Xmgk *)
(** {4 module Xmgk} *)
module Xmgk : sig
  type rgb = int * int * int
  type coord = Relative | Absolute
  type stroke = rgb * float
  type path_new = Path of coord * rgb option * stroke option

  type layers_closed

  type image_uninitialized
  type image
  type image_closed

  type prim =
    | New of int * int * rgb
    | Img of string
    | Layer of layers_closed

  val new_image : unit -> image_uninitialized
  val close_image : image -> image_closed

  val image_add_prim : prim -> image_uninitialized -> image

  type filter
  val image_add_filter : filter -> image -> image

  val display : filter
  val write : string -> filter

  type msl
  type msl_closed

  val new_msl : unit -> msl
  val close_msl : msl -> msl_closed

  val msl_add_image : image_closed -> msl -> msl
  val msl_print : msl_closed -> unit

  type path
  val new_path : path_new -> path
  val path_close : path -> filter

  type path_elem =
    | Move_to of int * int
    | Line_to of int * int
    | H_line_to of int
    | V_line_to of int
    | Quad_curve_to of (int * int) * (int * int)
    | Cubic_curve_to of (int * int) * (int * int) * (int * int)
    | Close_path

  val path_add_elem : path_elem -> path -> path

  val blur : float -> float -> filter
  val edge : float -> filter
  val negate : filter
  val charcoal : float -> float -> filter
  val modulate : int -> int -> int -> filter

  val emboss : float -> float -> filter
  val shade : float -> float -> filter

  val despeckle : filter
  val noise : float -> filter
  val roll : int -> int -> filter

  val brightness : int -> filter
  val saturation : int -> filter
  val hue : int -> filter

  val draw_rectangle : int * int -> int * int -> int * int * int -> filter
  val draw_circle : int * int -> int -> int * int * int -> filter

  (*
  type layed
  type primary

  type 'a layer
  type 'a layer_closed
  type layers_list
  *)
  (*
  type layer
  type primary_layer
  type layer_closed
  *)
  type layed
  type primary

  type 'a layer
  type 'a layer_closed
  type layers_list

  val new_layer_prim : unit -> primary layer
  val new_layer : string -> layed layer
  val close_layer : 'a layer -> 'a layer_closed

  val new_layers_list : primary layer_closed -> layers_list

  val layer_list_add_layer : layed layer_closed -> layers_list -> layers_list

  val layers_close : layers_list -> layers_closed

  val layer_add_prim : prim -> 'a layer -> 'a layer

  val layer_add_filter : filter -> 'a layer -> 'a layer

end
