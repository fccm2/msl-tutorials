    <merge>
      <layer>
        <new w='190' h='190' rgb='255,255,255' />
        <draw_rectangle pos='0,0' dims='50,70' rgb='0,0,255' />
      </layer>
      <layer compose='%[COMP]'>
        <new w='190' h='190' rgb='255,255,255' />
        <draw_rectangle pos='40,10' dims='40,40' rgb='255,0,0' />
      </layer>
    </merge>
  </image>
  <display />

</msl>
